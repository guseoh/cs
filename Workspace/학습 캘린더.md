# 학습 캘린더

[[Home|개발자 홈]] · [[Workspace/진행 보드|진행 보드]]

왼쪽 리본에서 **오늘의 일일 노트 열기**를 선택합니다. 달력의 날짜는 작성된 일일 노트에 연결됩니다. 진행 카드의 `review_on`과 `target_on`을 날짜 속성으로 지정하면 각각 복습과 마일스톤으로 표시됩니다.

```dataviewjs
const today = dv.date("today");
let month = today.startOf("month");
const calendar = dv.container.createDiv({ cls: "developer-calendar" });
const entries = new Map();

function add(day, entry) {
    if (!day?.isValid) return;
    const key = day.toISODate();
    if (!entries.has(key)) entries.set(key, []);
    entries.get(key).push(entry);
}

for (const page of dv.pages('"Workspace/일일 노트"')) {
    if (/^\d{4}-\d{2}-\d{2}$/.test(page.file.name)) {
        add(dv.date(page.file.name), { path: page.file.path, label: "일일 노트" });
    }
}

for (const page of dv.pages('"Workspace/Items"')) {
    for (const [property, label] of [["review_on", "복습"], ["target_on", "마일스톤"]]) {
        if (page[property]) add(dv.date(page[property]), { path: page.file.path, label: label + " · " + page.file.name });
    }
}

function render() {
    calendar.replaceChildren();
    const toolbar = calendar.createDiv({ cls: "developer-calendar-toolbar" });
    toolbar.createEl("strong", { text: month.setLocale("ko").toFormat("yyyy년 M월") });
    for (const [label, move] of [
        ["이전 달", () => month.minus({ months: 1 })],
        ["이번 달", () => today.startOf("month")],
        ["다음 달", () => month.plus({ months: 1 })],
    ]) {
        toolbar.createEl("button", { text: label }).addEventListener("click", () => {
            month = move();
            render();
        });
    }

    const grid = calendar.createDiv({ cls: "developer-calendar-grid" });
    for (const day of ["월", "화", "수", "목", "금", "토", "일"]) {
        grid.createDiv({ cls: "developer-calendar-weekday", text: day });
    }
    for (let blank = 1; blank < month.weekday; blank++) grid.createDiv();
    for (let number = 1; number <= month.daysInMonth; number++) {
        const day = month.set({ day: number });
        const cell = grid.createDiv({ cls: "developer-calendar-day" });
        if (day.hasSame(today, "day")) cell.classList.add("is-today");
        cell.createEl("span", { text: String(number) });
        for (const entry of entries.get(day.toISODate()) ?? []) {
            dv.el("div", dv.fileLink(entry.path, false, entry.label), { container: cell });
        }
    }
}

render();
```
