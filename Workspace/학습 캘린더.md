# 학습 캘린더

날짜를 선택하면 해당일의 일일 노트, 복습과 마일스톤을 아래에서 열 수 있습니다. 기록은 Core Daily Notes를, 일정 날짜는 작업 카드의 `review_on`과 `target_on`을 사용합니다.

```dataviewjs
const today = dv.date("today").startOf("day");
let month = today.startOf("month");
let selected = today;
const calendar = dv.container.createDiv({ cls: "developer-calendar" });
const entries = [];

function add(day, kind, path, label, source) {
    if (!day?.isValid) return;
    entries.push({ day: day.toISODate(), kind, path, label, source });
}

for (const page of dv.pages('"Workspace/일일 노트"')) {
    if (/^[0-9]{4}-[0-9]{2}-[0-9]{2}$/.test(page.file.name)) {
        add(dv.date(page.file.name), "기록", page.file.path, page.file.name);
    }
}

for (const page of dv.pages('"Workspace/Items"')) {
    const source = page.source_note ? dv.page(page.source_note)?.file.path : undefined;
    for (const [field, kind] of [["review_on", "복습"], ["target_on", "마일스톤"]]) {
        if (page[field]) add(dv.date(page[field]), kind, page.file.path, page.unit ?? page.file.name, source);
    }
}
entries.sort((a, b) => a.day.localeCompare(b.day) || a.label.localeCompare(b.label, "ko"));

function list(parent, title, rows, empty) {
    const section = parent.createEl("section");
    section.createEl("h3", { text: title });
    if (!rows.length) {
        section.createEl("p", { text: empty, cls: "developer-agenda-empty" });
    }
    for (const item of rows) {
        const row = section.createDiv({ cls: "developer-agenda-event" });
        row.createEl("span", { text: item.day.slice(5) + " · " + item.kind });
        dv.el("div", dv.fileLink(item.path, false, item.label), { container: row });
        if (item.source && item.source !== item.path) {
            dv.el("div", dv.fileLink(item.source, false, "원본 학습 자료"), { container: row });
        }
    }
}

function render() {
    calendar.replaceChildren();
    const toolbar = calendar.createDiv({ cls: "developer-calendar-toolbar" });
    toolbar.createEl("strong", { text: month.setLocale("ko").toFormat("yyyy년 M월") });
    for (const [label, next, nextDay] of [
        ["이전 달", month.minus({months: 1}), month.minus({months: 1})],
        ["이번 달", today.startOf("month"), today],
        ["다음 달", month.plus({months: 1}), month.plus({months: 1})],
    ]) {
        toolbar.createEl("button", { text: label }).addEventListener("click", () => {
            month = next; selected = nextDay; render();
        });
    }

    const grid = calendar.createDiv({ cls: "developer-calendar-grid" });
    for (const day of ["월", "화", "수", "목", "금", "토", "일"]) {
        grid.createDiv({ cls: "developer-calendar-weekday", text: day });
    }
    for (let blank = 1; blank < month.weekday; blank++) grid.createDiv();
    for (let number = 1; number <= month.daysInMonth; number++) {
        const day = month.set({ day: number });
        const iso = day.toISODate();
        const count = entries.filter((entry) => entry.day === iso).length;
        const cell = grid.createEl("button", { cls: "developer-calendar-day", text: String(number) });
        cell.setAttribute("aria-label", iso + (count ? " · " + count + "건" : ""));
        const active = selected.hasSame(day, "day");
        cell.setAttribute("aria-pressed", String(active));
        if (day.hasSame(today, "day")) cell.classList.add("is-today");
        if (count) cell.createEl("small", { text: count + "건" });
        cell.addEventListener("click", () => {
            selected = day; render();
            calendar.querySelector('[aria-pressed="true"]')?.focus();
        });
    }

    const panel = calendar.createDiv({ cls: "developer-agenda-items" });
    const weekStart = today.startOf("week").toISODate();
    const weekEnd = today.endOf("week").toISODate();
    list(panel, selected.toISODate() + " 기록과 일정", entries.filter((e) => e.day === selected.toISODate()),
        "선택한 날짜에 기록이나 일정이 없습니다.");
    list(panel, "이번 주 학습 기록", entries.filter((e) => e.kind === "기록" && e.day >= weekStart && e.day <= weekEnd),
        "이번 주에 작성한 일일 노트가 없습니다.");
    list(panel, "다가오는 복습과 마일스톤", entries.filter((e) => e.kind !== "기록" && e.day >= today.toISODate()).slice(0, 10),
        "작업 카드에 날짜를 입력하면 여기에 표시됩니다.");
}
render();
```
